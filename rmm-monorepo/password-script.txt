==========================================================================

generate api key used password from .env
===========================================================================
$email = 'admin@example.test'
$securePassword = Read-Host 'Bootstrap admin password' -AsSecureString
$password = [System.Net.NetworkCredential]::new('', $securePassword).Password

try {
  $body = @{ email = $email; password = $password } | ConvertTo-Json
  $login = Invoke-RestMethod `
    -Method Post `
    -Uri 'http://localhost:3002/auth/login' `
    -ContentType 'application/json' `
    -Body $body

  $env:HELP_DESK_API_KEY = $login.apiKey
  'API key generated and set for this PowerShell session.'
}
finally {
  Remove-Variable password, securePassword -ErrorAction SilentlyContinue
}


=========================================================================
to copy api key in same powershell window
=========================================================================
Set-Clipboard -Value $env:HELP_DESK_API_KEY
'API key copied to clipboard. Keep it private.'
